The test races the reaper: the child can exit before install_reaper() returns, so the SIGCHLD is lost. Here is the fix:

rust
pub fn install_reaper() -> io::Result<Reaper> {
    let mask = block_sigchld()?;
    let reaper = Reaper::new(mask)?;
    reaper.drain_exited();
    Ok(reaper)
}
